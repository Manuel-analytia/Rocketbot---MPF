   CMD07.e02 father=execScriptPython desc='Enviar alerta Excel no encontrado usando v_smtp_* y Para.txt'
      meta: {'run_when': 'True'}
      command:
        import smtplib
        from email.mime.text import MIMEText
        from email.mime.multipart import MIMEMultipart
        
        smtp_host = GetVar("v_smtp_host")
        smtp_port = int(GetVar("v_smtp_port") or 587)
        smtp_user = GetVar("v_smtp_user")
        smtp_pass = GetVar("v_smtp_pass")
        smtp_ssl  = GetVar("v_smtp_ssl").lower() == 'true'
        asunto    = GetVar("v_asunto_alerta")
        cuerpo    = GetVar("v_msg_alerta")
        
        # Leer destinatarios desde Para.txt
        ruta_para = GetVar("v_ruta_para_txt")
        destinatarios = []
        try:
            with open(ruta_para, 'r', encoding='utf-8') as f:
                for linea in f:
                    linea = linea.strip()
                    if linea and not linea.startswith('#'):
                        destinatarios.append(linea)
        except Exception:
            pass
        
        if not destinatarios or not smtp_host or not smtp_user:
            SetVar("v_email_config", "NO_CONFIG")
        else:
            msg = MIMEMultipart()
            msg['From'] = smtp_user
            msg['To']   = ', '.join(destinatarios)
            msg['Subject'] = asunto
            msg.attach(MIMEText(cuerpo, 'plain', 'utf-8'))
            try:
                if smtp_ssl:
                    server = smtplib.SMTP_SSL(smtp_host, smtp_port)
                else:
                    server = smtplib.SMTP(smtp_host, smtp_port)
                    server.starttls()
                server.login(smtp_user, smtp_pass)
                server.sendmail(smtp_user, destinatarios, msg.as_string())
                server.quit()
                SetVar("v_email_config", "ENVIADO")
            except Exception as e:
                SetVar("v_email_config", f"ERROR: {e}")
